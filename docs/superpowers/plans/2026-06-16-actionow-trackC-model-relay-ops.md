# Track C — Model-Relay Operations Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: superpowers:subagent-driven-development. Steps use `- [ ]` checkboxes. TDD.

**Goal:** Add the model-relay operating layer — a `model` registry (name→chat `resource`, public|tenant visibility), per-model token pricing (`model_token`) frozen at quote, redemption codes (`redeemCode` reusing `topUp`, idempotent on code), and per-call `usage_log` — with admin CRUD + global stats routes.

**Architecture:** New Drizzle tables (`model`, `model_grant`, `redemption_code`, `usage_log`) in `@actionow/db` with pure-SQL tenant-predicate repos (`resolveModelForTenant`, `listVisibleModels`, `insertUsageLog`); a new pricing method `'model_token'` added to `@actionow/contracts`/`@actionow/domain` priced at `inputTokens*inputPrice + outputTokens*outputPrice` µcredits and run through the existing `quote`/`settle` engine path; a `redeemCode` engine function reusing the `topUp` ledger path with code-level idempotency; and admin routes mirroring `AdminController`/`AdminService`.

**Tech Stack:** Drizzle ORM 0.45 + drizzle-kit 0.31 (Postgres), NestJS 11 ESM, zod, vitest + Testcontainers (`postgres:16-alpine`), supertest for e2e.

  ```ts
  import type { InferSelectModel } from 'drizzle-orm';
  import { and, desc, eq, inArray, or } from 'drizzle-orm';

  import { model, modelGrant } from '../schema/index.js';
  import type { DbOrTx } from './types.js';

  export type ModelRow = InferSelectModel<typeof model>;

  /** Fields a caller supplies to {@link createModel}. */
  export interface CreateModelInput {
    readonly name: string;
    readonly resourceId: string;
    readonly display?: string | null | undefined;
    readonly modality?: string | undefined;
    /** Feature flags (tools/vision/audio/json/stream) read by Track A. */
    readonly capabilities?: Record<string, boolean> | undefined;
    /** `public` | `tenant`. Defaults to `public` (db default). */
    readonly visibility?: string | undefined;
    /** µcredit-per-input-token integer (resale price). Defaults to 0. */
    readonly inputPrice?: bigint | undefined;
    /** µcredit-per-output-token integer. Defaults to 0. */
    readonly outputPrice?: bigint | undefined;
    readonly status?: string | undefined;
  }

  /** Insert a `model` row and return it. */
  export async function createModel(db: DbOrTx, input: CreateModelInput): Promise<ModelRow> {
    const rows = await db
      .insert(model)
      .values({
        name: input.name,
        resourceId: input.resourceId,
        ...(input.display !== undefined ? { display: input.display } : {}),
        ...(input.modality !== undefined ? { modality: input.modality } : {}),
        ...(input.capabilities !== undefined ? { capabilities: input.capabilities } : {}),
        ...(input.visibility !== undefined ? { visibility: input.visibility } : {}),
        ...(input.inputPrice !== undefined ? { inputPrice: input.inputPrice } : {}),
        ...(input.outputPrice !== undefined ? { outputPrice: input.outputPrice } : {}),
        ...(input.status !== undefined ? { status: input.status } : {}),
      })
      .returning();
    const row = rows[0];
    if (!row) throw new Error('createModel: insert returned no row');
    return row;
  }

  /** The shape Track A's relay needs to route + price a resolved model. */
  export interface ResolvedModel {
    readonly modelId: string;
    readonly resourceId: string;
    readonly inputPrice: bigint;
    readonly outputPrice: bigint;
    readonly capabilities: Record<string, boolean>;
  }

  /**
   * Resolve a model `name` for `tenantId` (§C.1 visibility gate). Returns the
   * routing + pricing handle when the model is `active` AND either `public` or
   * `tenant`-visible WITH a `model_grant` for this tenant; otherwise `undefined`
   * (an ungranted tenant model is indistinguishable from a missing one). The
   * tenant predicate is in the SQL `WHERE`, never a read-then-check.
   */
  export async function resolveModelForTenant(
    db: DbOrTx,
    modelName: string,
    tenantId: string,
  ): Promise<ResolvedModel | undefined> {
    const rows = await db
      .select({
        id: model.id,
        resourceId: model.resourceId,
        inputPrice: model.inputPrice,
        outputPrice: model.outputPrice,
        capabilities: model.capabilities,
        visibility: model.visibility,
        grantTenant: modelGrant.tenantId,
      })
      .from(model)
      .leftJoin(
        modelGrant,
        and(eq(modelGrant.modelId, model.id), eq(modelGrant.tenantId, tenantId)),
      )
      .where(and(eq(model.name, modelName), eq(model.status, 'active')))
      .limit(1);

    const row = rows[0];
    if (row === undefined) return undefined;
    if (row.visibility !== 'public' && row.grantTenant === null) return undefined;

    return {
      modelId: row.id,
      resourceId: row.resourceId,
      inputPrice: row.inputPrice,
      outputPrice: row.outputPrice,
      capabilities: row.capabilities,
    };
  }

  /** A model as seen by a tenant (console `GET /console/models` / discovery). */
  export interface ModelView {
    readonly id: string;
    readonly name: string;
    readonly display: string | null;
    readonly modality: string;
    readonly capabilities: Record<string, boolean>;
    readonly inputPrice: bigint;
    readonly outputPrice: bigint;
  }

  /**
   * List the models visible to `tenantId` (§C.1): every `active` `public` model,
   * plus every `active` `tenant` model this tenant has a `model_grant` for.
   * Consumed by Track B's `GET /console/models`.
   */
  export async function listVisibleModels(db: DbOrTx, tenantId: string): Promise<ModelView[]> {
    const granted = await db
      .select({ modelId: modelGrant.modelId })
      .from(modelGrant)
      .where(eq(modelGrant.tenantId, tenantId));
    const grantedIds = granted.map((g) => g.modelId);

    const visibility =
      grantedIds.length === 0
        ? eq(model.visibility, 'public')
        : or(eq(model.visibility, 'public'), inArray(model.id, grantedIds));

    const rows = await db
      .select({
        id: model.id,
        name: model.name,
        display: model.display,
        modality: model.modality,
        capabilities: model.capabilities,
        inputPrice: model.inputPrice,
        outputPrice: model.outputPrice,
      })
      .from(model)
      .where(and(eq(model.status, 'active'), visibility))
      .orderBy(desc(model.createdAt));

    return rows;
  }

  /** List EVERY model newest-first (admin catalog view, incl. disabled). */
  export async function listModels(db: DbOrTx): Promise<ModelRow[]> {
    return db.select().from(model).orderBy(desc(model.createdAt));
  }

  /** Fetch a model by `name`, or `undefined` (admin lookups by the unique name). */
  export async function getModelByName(db: DbOrTx, name: string): Promise<ModelRow | undefined> {
    const rows = await db.select().from(model).where(eq(model.name, name)).limit(1);
    return rows[0];
  ```

  ```ts
      .from(redemptionCode)

  /** The disjoint outcomes of {@link claimRedemptionCode}. */
  export interface ClaimRedemptionResult {
    /** `true` iff THIS call flipped unused→redeemed (caller should credit). */
    readonly claimed: boolean;
    /** The µcredit grant, present when `claimed` is `true`. */
    readonly creditAmount?: bigint;
    /** Set when the code was already redeemed (the redeeming tenant). */
    readonly alreadyRedeemedBy?: string;
    /** Set when no code with that value exists. */
    readonly missing?: boolean;
    /** Set when the code exists but is past `expires_at`. */
    readonly expired?: boolean;
  }

  /**
   * Atomically claim `code` for `tenantId` (§C.3). A single conditional UPDATE
   * (`WHERE status='unused' AND (expires_at IS NULL OR expires_at > now())
   * RETURNING credit_amount`) flips the row — there is no read-then-write window,
   * so concurrent claims cannot double-redeem. When the UPDATE matches nothing we
   * read the row back to disambiguate `missing` / `expired` / already-redeemed, so
   * the engine `redeemCode` no-ops idempotently on a re-claim.
   */
  export async function claimRedemptionCode(
    db: DbOrTx,
    code: string,
    tenantId: string,
  ): Promise<ClaimRedemptionResult> {
    const updated = await db
      .update(redemptionCode)
      .set({ status: 'redeemed', redeemedByTenant: tenantId, redeemedAt: sql`now()` })
      .where(
        and(
          eq(redemptionCode.code, code),
          eq(redemptionCode.status, 'unused'),
          sql`(${redemptionCode.expiresAt} IS NULL OR ${redemptionCode.expiresAt} > now())`,
  ```

  ```ts
  import { claimRedemptionCode, type DbOrTx, getAccountByTenant, topUp } from '@actionow/db';

  import { engineError } from './errors.js';

  /** The outcome of {@link redeemCode}. */
  export interface RedeemResult {
    /** `true` iff THIS call claimed the code and credited the account. */
    redeemed: boolean;
    /** The post-credit balance when `redeemed` is `true`. */
    balance?: bigint;
  }

  /**
   * Redeem a code for a tenant (§C.3): atomically claim the code, then — only when
   * THIS call won the claim — credit the tenant's account via the existing `topUp`
   * ledger path (one `topup` ledger row). Idempotent on the code: a re-claim
   * returns `{redeemed:false}` and credits nothing (the claim UPDATE matched no
   * row). Throws `code_not_found` for an unknown code, `code_expired` for an
   * expired one, and `internal_error` when the tenant has no account.
   *
   * The console redeem route (Track B) calls this with the JWT-resolved tenant.
   */
  export async function redeemCode(
    db: DbOrTx,
    code: string,
    tenantId: string,
  ): Promise<RedeemResult> {
    const claim = await claimRedemptionCode(db, code, tenantId);

    if (!claim.claimed) {
      if (claim.missing === true) {
        throw engineError('resource_not_found', `code ${code} not found`);
      }
      if (claim.expired === true) throw engineError('invalid_input', `code ${code} expired`);
      // Already redeemed (by this or another tenant) → idempotent no-op.
      return { redeemed: false };
    }

    if (claim.creditAmount === undefined) {
      throw engineError('internal_error', 'claimed code missing credit_amount');
    }

    const account = await getAccountByTenant(db, tenantId);
    if (account === undefined) {
      throw engineError('internal_error', `no account for tenant ${tenantId}`);
    }

    const balance = await topUp(db, account.id, tenantId, claim.creditAmount);
    return { redeemed: true, balance };
  }
  ```
